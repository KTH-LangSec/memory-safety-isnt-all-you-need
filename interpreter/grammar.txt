n  ::= '0' | '1' | '2' | ...                                           # ^1
x  ::= '_' | 'a' | 'b' | 'c' | ... | '_1' | ...                        # ^1
lo ::= `0 | `1 | `2 | ...                                              # ^1
lf ::= `0f | `1f | `2f | ...                                           # ^1
l  ::= <lo> | <lf>                                                     # ^1
op ::= '+' | '-' | '&&' | '==' | '!='                                  # ^2
v  ::= 'nil' | <n> | <l>                                               # ^1, ^3
e  ::= 'nil'
     | <n>
     | <x>
     | <e1> <op> <e2>
     | '(' <e> ')'                                                     # ^1
c  ::= 'skip'
     | <c1> ';' <c2>
     | <x> ':=' <e>
     | <x> '[' <e_i> ']' ':=' <e>
     | <x> ':=' <x_o> '[' <e_i> ']'
     | <x> '<-' 'alloc'
     | <x> '<-' 'alloc_C'                                              # ^1
     | 'fn' <x> '{' <c> '}'
     | 'call' <x>
     | 'if' <e> '{' <c1> '}' 'else' '{' <c2> '}'
     | 'while' <e> '{' <c> '}'
